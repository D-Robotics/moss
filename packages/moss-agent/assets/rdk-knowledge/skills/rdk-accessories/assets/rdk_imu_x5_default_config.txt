# RDK_IMU_X5_DEFAULT_CONFIG — official default config for the RDK IMU Module (BMI088) on RDK X5.
# Source: accessories_doc docs/03_imu_module/05_software/02_c_api.md.
# Use as the starting point for rdk_imu_config_t; override only the members you need.
# In C: `rdk_imu_config_t config = RDK_IMU_X5_DEFAULT_CONFIG;` then change fields.
# The same values map 1:1 to the ROS2 node parameters (see references/accessories-catalog.md).

# --- Members set verbatim by the official RDK_IMU_X5_DEFAULT_CONFIG macro ---
accel_drdy_gpio_chip  = 4                 # SoC-side GPIO chip receiving the accel interrupt
accel_drdy_gpio_line  = 2                 # SoC-side GPIO line
gyro_drdy_gpio_chip   = 3                 # SoC-side GPIO chip receiving the gyro interrupt
gyro_drdy_gpio_line   = 12                # SoC-side GPIO line
accel_range           = RDK_IMU_ACCEL_24G # +/- 24 g
accel_odr             = RDK_IMU_ACCEL_400 # 400 Hz
gyro_range            = RDK_IMU_GYRO_2000DPS  # +/- 2000 dps
gyro_bandwidth        = RDK_IMU_ODR400_BW47   # gyro ODR 400 Hz / bandwidth 47 Hz (bound pair)
fifo_length           = 256              # software FIFO length (must be a power of two)
fifo_mode             = RDK_IMU_FIFO_OVERWRITE  # overwrite oldest when full (vs DROP)
